     7→async function api(path: string, opts: RequestInit = {}) {
     8→  const token = typeof localStorage !== 'undefined' ? localStorage.getItem('session_token') || '' : '';
     9→  const res = await fetch(path, {
    10→    ...opts,
    11→    headers: { 'Content-Type': 'application/json', 'x-session-token': token, ...(opts.headers as Record<string, string> || {}) },
    12→  });
    13→  if (res.status === 401) { if (_onUnauthorized) _onUnauthorized(); throw new Error('unauthorized'); }
    14→  const data = await res.json().catch(() => ({}));
    15→  if (!res.ok) throw new Error((data as any).error || `HTTP ${res.status}`);
    16→  return data;
    17→}

    19→function timeAgo(ts: number | string) {
    20→  if (!ts) return 'never';
    21→  const d = new Date(ts);
    22→  const sec = Math.floor((Date.now() - d.getTime()) / 1000);
    23→  if (sec < 60) return sec + 's ago';
    24→  if (sec < 3600) return Math.floor(sec / 60) + 'm ago';
    25→  if (sec < 86400) return Math.floor(sec / 3600) + 'h ago';
    26→  return d.toLocaleDateString();
    27→}

    29→function esc(s: any) {
    30→  return String(s ?? '').replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[c]);
    31→}

    63→function Login({ onLogin }: { onLogin: () => void }) {
    64→  const [pw, setPw] = useState(''); const [err, setErr] = useState(''); const [loading, setLoading] = useState(false);
    65→  const submit = async (e: React.FormEvent) => {
    66→    e.preventDefault(); setErr(''); setLoading(true);
    67→    try {
    68→      const r = await fetch('/api/auth/login', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ password: pw }) });
    69→      const data = await r.json();
    70→      if (!r.ok) throw new Error(data.error || 'login failed');
    71→      localStorage.setItem('session_token', data.token);
    72→      onLogin();
    73→    } catch (e) { setErr((e as Error).message); } finally { setLoading(false); }
    74→  };
    75→  return (
    76→    <div className="min-h-screen flex items-center justify-center p-4 bg-background">
    77→      <div className="w-full max-w-md bg-card border border-border rounded-2xl p-8 shadow-xl">
    78→        <div className="flex items-center gap-3 mb-6">
    79→          <div className="w-12 h-12 rounded-xl bg-primary/15 flex items-center justify-center text-2xl">🤖</div>
    80→          <div><h1 className="text-xl font-bold">Bot Fleet</h1><p className="text-xs text-muted-foreground">Multi-bot Discord control panel</p></div>
    81→        </div>
    82→        <form onSubmit={submit} className="space-y-4">
    83→          <input type="password" value={pw} onChange={e => setPw(e.target.value)} required autoFocus placeholder="Admin password"
    84→            className="w-full bg-background border border-border rounded-lg px-3 py-2 text-sm" />
    85→          <button type="submit" disabled={loading} className="w-full bg-primary text-primary-foreground rounded-lg py-2 text-sm font-medium disabled:opacity-50">
    86→            {loading ? 'Signing in…' : 'Sign in'}
    87→          </button>
    88→          {err && <div className="text-xs text-red-500">{err}</div>}
    89→        </form>
    90→      </div>
    91→    </div>
    92→  );
    93→}

   168→// ─── Bot Editor ───────────────────────────────────────────────────────────
   169→function BotEditor({ bot, allBots, onClose, onSaved }: { bot: Bot | null; allBots: Bot[]; onClose: () => void; onSaved: () => void }) {
   170→  const isCreate = !bot;
   171→  const [name, setName] = useState('');
   172→  const [persona, setPersona] = useState('');
   173→  const [token, setToken] = useState('');
   174→  const [guildId, setGuildId] = useState('');
   175→  const [channelIds, setChannelIds] = useState<string[]>([]);
   176→  const [providers, setProviders] = useState<Provider[]>([]);
   177→  const [prob, setProb] = useState(1.0);
   178→  const [skipPats, setSkipPats] = useState('^\\+$\n^-$\n^(k|kk)$');
   179→  const [cooldown, setCooldown] = useState(2000);
   180→  const [maxCtx, setMaxCtx] = useState(1000000);
   181→  const [delay, setDelay] = useState(0);
   182→  const [ignoreBots, setIgnoreBots] = useState(false);
   183→  const [tools, setTools] = useState({ ...DEFAULT_TOOLS });
   184→  const [delegatedBots, setDelegatedBots] = useState<string[]>([]);
   185→  const [guilds, setGuilds] = useState<any[]>([]);
   186→  const [chLoading, setChLoading] = useState(false);
   187→  const [error, setError] = useState('');
   188→  const [saving, setSaving] = useState(false);
   189→
   190→  useEffect(() => {
   191→    if (!bot) return;
   192→    api(`/api/bots/${bot.id}`).then((b: any) => {
   193→      setName(b.name || ''); setPersona(b.persona || ''); setGuildId(b.guild_id || '');
   194→      setChannelIds(b.channel_ids || []); setProviders(b.providers || []);
   195→      setProb(b.gating?.response_probability ?? 1.0);
   196→      setSkipPats((b.gating?.skip_patterns || ['^\\+$', '^-$', '^(k|kk)$']).join('\n'));
   197→      setCooldown(b.gating?.cooldown_ms ?? 2000);
   198→      setMaxCtx(b.gating?.max_context_messages ?? 1000000);
   199→      setDelay(b.gating?.response_delay_ms ?? 0);
   200→      setIgnoreBots(b.gating?.ignore_bots ?? false);
   201→      setTools({ ...DEFAULT_TOOLS, ...(b.tools || {}) });
   202→      setDelegatedBots(b.delegated_bots || []);
   203→    }).catch((e: any) => setError(e.message));
   204→    // Auto-load channels
   205→    api(`/api/bots/${bot.id}/channels`).then((data: any) => {
   206→      if (data.guilds) setGuilds(data.guilds);
   207→    }).catch(() => {});
   208→  }, [bot]);
   209→
   210→  const loadChannels = async () => {
   211→    setChLoading(true); setError('');
   212→    try {
   213→      let data: any;
   214→      if (token) {
   215→        data = await api('/api/discord/guilds', { method: 'POST', body: JSON.stringify({ token }) });
   216→      } else if (bot) {
   217→        data = await api(`/api/bots/${bot.id}/channels`);
   218→      } else { setError('Paste token to load channels'); return; }
   219→      setGuilds(data.guilds || []);
   220→    } catch (e) { setError((e as Error).message); }
   221→    finally { setChLoading(false); }
   222→  };
   223→
   224→  const save = async (restart: boolean) => {
   225→    setError(''); setSaving(true);
   226→    const body: any = {
   227→      name, persona, guild_id: guildId, channel_ids: channelIds, delegated_bots: delegatedBots,
   228→      providers: providers.map(p => {
   229→        const out: any = { ...p };
   230→        delete out.api_key_masked;
   231→        return out;
   232→      }),
   233→      gating: { response_probability: prob, skip_patterns: skipPats.split('\n').map(s => s.trim()).filter(Boolean), ignore_bots: ignoreBots, max_context_messages: maxCtx, cooldown_ms: cooldown, response_delay_ms: delay },
   234→      tools,
   235→    };
   236→    if (token) body.token = token;
   237→    try {
   238→      if (bot) {
   239→        await api(`/api/bots/${bot.id}`, { method: 'PUT', body: JSON.stringify(body) });
   240→        if (restart) await api(`/api/bots/${bot.id}/restart`, { method: 'POST' });
   241→      } else {
   242→        await api('/api/bots', { method: 'POST', body: JSON.stringify(body) });
   243→      }
   244→      onSaved(); onClose();
   245→    } catch (e) { setError((e as Error).message); }
   246→    finally { setSaving(false); }
   247→  };
   248→
   249→  return (
   250→    <div className="fixed inset-0 z-50 bg-black/70 overflow-y-auto" onClick={onClose}>
   251→      <div className="bg-card border border-border rounded-2xl w-full max-w-3xl mx-auto my-4 min-h-min" onClick={e => e.stopPropagation()} style={{ maxWidth: '900px' }}>
   252→        <div className="px-6 py-3 border-b border-border flex items-center justify-between sticky top-0 bg-card z-10 rounded-t-2xl">
   253→          <h2 className="font-semibold text-sm">{isCreate ? 'Create New Bot' : `Edit: ${esc(name)}`}</h2>
   254→          <button onClick={onClose} className="text-muted-foreground hover:text-foreground text-xl">×</button>
   255→        </div>
   256→        <div className="p-6 space-y-5">
   257→          {/* Name */}
   258→          <div><label className="block text-xs text-muted-foreground mb-1">Name *</label>
   259→            <input value={name} onChange={e => setName(e.target.value)} placeholder="e.g. Scylla, Mama, Narrator" className="w-full bg-background border border-border rounded-lg px-3 py-2 text-sm" /></div>
   260→          {/* Persona */}
   261→          <div><label className="block text-xs text-muted-foreground mb-1">Persona (system prompt) *</label>
   262→            <textarea value={persona} onChange={e => setPersona(e.target.value)} rows={8} placeholder="You are..." className="w-full bg-background border border-border rounded-lg px-3 py-2 text-sm font-mono" /></div>
   263→          {/* Token + Load channels */}
   264→          <div><label className="block text-xs text-muted-foreground mb-1">Discord token {isCreate ? '*' : '(blank = keep existing)'}</label>
   265→            <div className="flex gap-2">
   266→              <input type="password" value={token} onChange={e => setToken(e.target.value)} placeholder="MTUzOTE5..." className="flex-1 bg-background border border-border rounded-lg px-3 py-2 text-sm font-mono" />
   267→              <button onClick={loadChannels} disabled={chLoading} className="text-xs bg-primary text-primary-foreground rounded-lg px-3 py-2 whitespace-nowrap disabled:opacity-50">{chLoading ? '...' : '→ Load channels'}</button>
   268→            </div></div>
   269→          {/* Guild ID */}
   270→          <div><label className="block text-xs text-muted-foreground mb-1">Guild ID</label>
   271→            <input value={guildId} onChange={e => setGuildId(e.target.value)} placeholder="1511640846435356794" className="w-full bg-background border border-border rounded-lg px-3 py-2 text-sm font-mono" /></div>
   272→          {/* Channels */}
   273→          {guilds.length > 0 && (
   274→            <div><label className="block text-xs text-muted-foreground mb-1">Channels (empty = all)</label>
   275→              <div className="max-h-48 overflow-y-auto bg-background border border-border rounded-lg p-2 space-y-1">
   276→                {guilds.map(g => (
   277→                  <div key={g.id}>
   278→                    <div className="text-xs font-semibold text-foreground mt-1 mb-1">{esc(g.name)}</div>
   279→                    {(g.text_channels || []).map((c: any) => (
   280→                      <label key={c.id} className="flex items-center gap-2 text-xs py-0.5 cursor-pointer hover:bg-muted/30 px-1 rounded">
   281→                        <input type="checkbox" checked={channelIds.includes(c.id)} onChange={e => {
   282→                          setChannelIds(e.target.checked ? [...channelIds, c.id] : channelIds.filter(id => id !== c.id));
   283→                          if (e.target.checked && !guildId) setGuildId(g.id);
   284→                        }} className="accent-primary" />
   285→                        <span>#{esc(c.name)}</span>
   286→                        <span className="text-muted-foreground/50 font-mono ml-auto text-[10px]">{c.id}</span>
   287→                      </label>
   288→                    ))}
   289→                  </div>
   290→                ))}
   291→              </div></div>
   292→          )}
   293→          {/* Providers */}
   294→          <div className="border-t border-border pt-4">
   295→            <div className="flex items-center justify-between mb-2">
   296→              <h3 className="font-semibold text-sm">Providers (racing)</h3>
   297→              <button onClick={() => setProviders([...providers, { id: 'p' + Date.now(), name: 'New Provider', type: 'openai', priority: providers.length + 1, enabled: true, proxy_url: '', model: '', temperature: 0.85, max_tokens: 1500 }])} className="text-xs bg-primary text-primary-foreground rounded px-2 py-1">+ Add</button>
   298→            </div>
   299→            <div className="space-y-2">
   300→              {providers.map((p, i) => (
   301→                <ProviderEditor key={p.id} p={p} onChange={np => setProviders(providers.map((x, idx) => idx === i ? np : x))} onRemove={() => setProviders(providers.filter((_, idx) => idx !== i))} />
   302→              ))}
   303→              {providers.length === 0 && <p className="text-xs text-muted-foreground">No providers. Add one to get started.</p>}
   304→            </div>
   305→          </div>
   306→          {/* Gating */}
   307→          <div className="border-t border-border pt-4">
   308→            <h3 className="font-semibold text-sm mb-2">Behavior</h3>
   309→            <div className="space-y-3">

   311→                <input type="range" min="0" max="1" step="0.05" value={prob} onChange={e => setProb(parseFloat(e.target.value))} className="w-full accent-primary" /></div>
   312→              <div><label className="text-xs text-muted-foreground">Skip patterns (one per line)</label>

   316→                <label className="text-xs">Max context<input type="number" value={maxCtx} onChange={e => setMaxCtx(parseInt(e.target.value))} className="w-full bg-background border border-border rounded px-2 py-1 text-sm" /></label>

   318→              </div>

   328→            <div className="grid grid-cols-2 gap-2">
   329→              {Object.keys(TOOL_DESCS).map(k => (
   330→                <label key={k} className="flex items-center justify-between bg-background border border-border rounded-lg px-3 py-2 cursor-pointer hover:bg-muted/30">
   331→                  <span className="text-sm">{TOOL_DESCS[k]}</span>
   332→                  <input type="checkbox" checked={tools[k] ?? false} onChange={e => setTools({ ...tools, [k]: e.target.checked })} className="accent-primary" />
   333→                </label>
   334→              ))}
   335→            </div>
   336→          </div>
   337→          {/* Delegated bots */}
   338→          {allBots.length > 1 && (
   339→            <div className="border-t border-border pt-4">
   340→              <h3 className="font-semibold text-sm mb-2">Delegated bots (can summon)</h3>
   341→              <div className="max-h-32 overflow-y-auto bg-background border border-border rounded-lg p-2 space-y-1">
   342→                {allBots.filter(b => !bot || b.id !== bot.id).map(other => (
   343→                  <label key={other.id} className="flex items-center gap-2 text-xs cursor-pointer hover:bg-muted/30 px-1 rounded">

   345→                    <span>{esc(other.name)}</span>
   346→                  </label>
   347→                ))}
   348→              </div>
   349→            </div>
   350→          )}
   351→          {error && <div className="text-xs text-red-500 bg-red-500/10 p-2 rounded">{error}</div>}
   352→        </div>
   353→        <div className="px-6 py-3 border-t border-border flex gap-2 sticky bottom-0 bg-card rounded-b-2xl">
   354→          <button onClick={() => save(false)} disabled={saving} className="flex-1 bg-primary text-primary-foreground rounded-lg py-2 text-sm font-medium disabled:opacity-50">{saving ? 'Saving...' : 'Save'}</button>
   355→          {bot && <button onClick={() => save(true)} disabled={saving || bot.status !== 'running'} className="bg-muted border border-border rounded-lg px-4 py-2 text-sm disabled:opacity-50">Save & Restart</button>}
   356→          <button onClick={onClose} className="bg-muted border border-border rounded-lg px-4 py-2 text-sm">Cancel</button>
   357→        </div>
   358→      </div>
   359→    </div>
   360→  );

   390→// ─── Logs Tab ──────────────────────────────────────────────────────────────
   391→function LogsTab({ bots }: { bots: Bot[] }) {
   392→  const [botId, setBotId] = useState('');
   393→  const [logs, setLogs] = useState<string[]>([]);
   394→  const [auto, setAuto] = useState(false);
   395→  const ref = useRef<HTMLPreElement>(null);
   396→  useEffect(() => { if (bots.length && !botId) setBotId(bots[0].id); }, [bots]);
   397→  useEffect(() => {
   398→    if (!botId) return;
   399→    const fetchLogs = async () => {
   400→      await new Promise(r => setTimeout(r, 0));
   401→      try { const r = await api(`/api/bots/${botId}/logs?lines=200`); setLogs((r as any).logs || []); }
   402→      catch {}
   403→    };
   404→    fetchLogs();
   405→    if (auto) { const t = setInterval(fetchLogs, 3000); return () => clearInterval(t); }
   406→  }, [botId, auto]);
   407→  useEffect(() => { if (ref.current) ref.current.scrollTop = ref.current.scrollHeight; }, [logs]);
   408→  return (
   409→    <div className="space-y-3">
   410→      <div className="flex gap-2 items-center">
   411→        <select value={botId} onChange={e => setBotId(e.target.value)} className="bg-background border border-border rounded-lg px-3 py-2 text-sm">
   412→          {bots.map(b => <option key={b.id} value={b.id}>{esc(b.name)}</option>)}
   413→        </select>
   414→        <label className="text-xs flex items-center gap-1"><input type="checkbox" checked={auto} onChange={e => setAuto(e.target.checked)} className="accent-primary" /> Auto-refresh</label>
   415→      </div>
   416→      <pre ref={ref} className="bg-black/40 border border-border rounded-lg p-3 text-xs font-mono h-[600px] overflow-y-auto whitespace-pre-wrap">
   417→{logs.map(l => { try { const j = JSON.parse(l); return `[${j.ts?.slice(11, 19)}] ${j.level}: ${j.msg}`; } catch { return l; } }).join('\n')}
   418→      </pre>
   419→    </div>
   420→  );
   421→}

   448→
   449→  if (!loggedIn) return <Login onLogin={() => { setLoggedIn(true); refresh(); }} />;

   465→          </div>
   466→        </div>

   484→        {tab === 'logs' && <LogsTab bots={bots} />}
   485→      </main>

   487→      {(editing || creating) && (
   488→        <BotEditor bot={editing} allBots={bots} onClose={() => { setEditing(null); setCreating(false); }} onSaved={refresh} />
   489→      )}
   490→    </div>
